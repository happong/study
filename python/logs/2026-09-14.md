# Q6
def find_first_error(logs):
    for log in logs:
        if log == "에러":
            return "발견"
    return "없음"
logs = ["정보", "경고", "에러", "정보"]
print(find_first_error(logs))   # 발견
# ✅ (1차에 스스로 진단 및 수정 정확)

# Q7
def filter_warm(temps):
    higt = []
    for num in temps:
        if num >= 20:
            higt.append(num)
    return higt
temps = [18,25,30,12,27]
print(filter_warm(temps))   # [25, 30, 27]
# ✅ (1차에 정답)

# Q8 (1회 수정: 반복마다 maxplay 덮어쓰기 + user/longUser 대입 방향 반대 → 반복문 밖 초기화 + 방향 수정)
def most_active(play_logs):
    longUser = "a"
    maxplay = 0
    for user, playtime in play_logs:
        if playtime > maxplay:
           maxplay = playtime
           longUser = user
    return longUser
play_logs = [("하은", 120), ("민수", 340), ("지혜", 90)]
print(most_active(play_logs))   # 민수
# ✅

# Q9
def get_count(inventory, item_name):
    if item_name in inventory:
        return inventory[item_name]
    else:
        return 0
inventory = {"물약": 5, "두루마리": 2}
print(get_count(inventory, "물약"))   # 5
print(get_count(inventory, "검"))     # 0
# ✅ (1차에 정답)

# Q10 (3회 수정: 리스트 전체를 튜플처럼 언패킹 시도 → ValueError,
#      매출을 문자열 + 로 이어붙이려 시도, salesList에 salesList 자신을 append하는 오타(자기참조 리스트, 에러 없이 이상한 값) →
#      최종적으로 for로 하나씩 언패킹 + sales를 append + min/max 튜플 반환으로 완성)
def sales_range(daily_reprots):
    salesList = []
    for date, sales in daily_reprots:
        salesList.append(sales)
    return min(salesList), max(salesList)
daily_reports = [("09-10", 1200), ("09-11", 800), ("09-12", 2500)]
print(sales_range(daily_reports))   # (800, 2500)
# ✅

### 새로 발견된 실수 패턴

- **Q2**: 딕셔너리 키(문자열)를 따옴표 없이 변수 이름처럼 씀 (`config[theme]`) → `NameError`. 이전에 봤던 "매개변수 대신 리터럴 사용"의 반대 케이스 — 리터럴 문자열이어야 하는데 변수처럼 씀
- **Q5**: 문자열 조합 시 구분자 공백, 접미사(`원`) 등 세부 포맷 — 로직은 맞지만 완성도가 부족한 패턴이 반복됨 (2026-09-09에도 유사)
- **Q8**: Lv.10~11에서 반복된 "최댓값 찾기" 초기화/갱신 위치 실수 — 이번엔 스스로 한 번에 정확히 고침 (실력 향상 확인)
- **Q10**: 비슷한 이름의 변수(`sales` vs `salesList`)를 착각해서 리스트에 자기 자신을 append → 에러 없이 자기참조(self-referential) 리스트가 되어 이상한 출력. **에러가 안 나는 버그가 에러 나는 버그보다 위험할 수 있다**는 걸 보여준 사례
