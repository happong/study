def find_boss(monsters):
    result = "없음"
    for name, type_ in monsters:
        if type_ == "보스":
            result = name
    return result
monsters = [("슬라임", "일반"), ("드래곤", "보스"), ("고블린", "일반")]
print(find_boss(monsters))

def count_big_hits(damages):
    num = 0
    for count in damages:
        if count >= 50:
            num = num +1
    return num
damages = [15,80,45,120,30]
print(count_big_hits(damages))
# ✅ 2

# Q8
def save_guild_report(contributions):
    with open("guild_report.txt","w") as f:
        for name, percent in contributions.items():
            f.write(name+": "+str(percent)+"\n")
    with open("guild_report.txt","r") as f:
        return f.read()
contributions = {"하은":320,"민수":150,"지혜":480}
print(save_guild_report(contributions))
# ✅

# Q9
def write_crash_log(logs):
    with open("crash_log.txt","w") as f:
        for log in logs:
            f.write(log+"\n")
    with open("crash_log.txt","r") as f:
        listnum = f.readlines()
        return len(listnum)
logs = ["null 참조 오류", "메모리 초과", "타임아웃"]
print(write_crash_log(logs))
# ✅ 3

# Q10
def save_best_seller(sales):
    with open("best_seller.txt","w") as f:
        bestItem = "a"
        bestCount = 0
        for itemN, count in sales:
            if bestCount < count:
                bestCount = count
                bestItem = itemN
        f.write(bestItem)
    with open("best_seller.txt","r")as f:
        return f.read()
sales = [("물약", 300), ("장검", 45), ("방어구 세트", 120)]
print(save_best_seller(sales))
# ✅ 물약
