# Q7 (1회 수정: range(waits) → waits 그대로 순회)
def filter_long_wait(waits):
    numList = []
    for num in waits:
        if num >= 30:
            numList.append(num)
    return numList
waits = [12, 45, 8, 60, 30]
print(filter_long_wait(waits))   # [45, 60, 30]
# ✅

# Q8 (로직은 1차부터 정확, 본인이 테스트 데이터를 바꾸면서 주석만 안 고친 해프닝)
def most_severe(bug_reprots):
    securityMax = 0
    maxId = "a"
    for bugId, security in bug_reprots:
        if security >= securityMax:
            securityMax = security
            maxId = bugId
    return maxId
bug_reports = [("BUG-1", 3), ("BUG-2", 9), ("BUG-3", 5)]   # 원래 데이터로 검증
print(most_severe(bug_reports))   # BUG-2
# ✅ (원래 문제의 최댓값 추적 로직 완벽하게 적용함 — Lv.11 반복 학습 효과 확인)

# Q9
def get_price(item_prices, name):
    for key, price in item_prices.items():
        if key == name:
            return price
    return -1
item_prices = {"검": 3000, "방패": 5000}
print(get_price(item_prices, "검"))     # 3000
print(get_price(item_prices, "지팡이"))  # -1
# ✅ (1차 정답)

# Q10 (1회 수정: for 안에서 return min(time),max(time) 시도(int 하나로는 불가) → times 리스트에 모은 뒤 min/max)
def time_range(session_logs):
    times = []
    for userId, time in session_logs:
        times.append(time)
    return min(times), max(times)
session_logs = [("u1", 15), ("u2", 42), ("u3", 7)]
print(time_range(session_logs))   # (7, 42)
# ✅

### 새로 발견된 실수 패턴

- **Q7**: `range()`는 정수를 받아 "0부터 그 수 전까지"를 만드는 함수인데, 리스트를 그대로 `range(리스트)`에 넣어 순회하려 함 → `TypeError: 'list' object cannot be interpreted as an integer`. 리스트를 그냥 순회할 땐 `range()` 없이 `for x in 리스트:`로 충분함
- **Q8 (코드 결함 아님, 기록용)**: 함수 로직은 1차부터 정확했음. 테스트 데이터를 스스로 바꾸면서 주석(예상값)은 갱신하지 않아 결과가 "틀린 것처럼" 보였던 해프닝 — 실제로는 정답이었음. 테스트 데이터를 바꿀 땐 예상값 주석도 함께 확인하는 습관 참고
