### 제출 코드

```python
# 초급
note = {"title":"회의록","pages":5}
print(note["pages"])

settings = {"volume":50}
settings["brightness"] = 80
print(settings)

menu = {"drink":"커피","food":"샌드위치"}
print(menu["food"])

player = {"name":"라이언"}
if "skill" in player:
    print("있음")
else:
    print("없음")

# 중급
stock = {"사과":10,"바나나":0,"포도":5}
for fruit,num in stock.items():
    print(fruit, ":",num)

task = {"name":"버그 수정","status":"진행중"}
task["status"]="완료"
print(task)

def is_urgent(ticket_dict):
   if ticket_dict["priority"] == "긴급":
       return True
   else:
       return False

# 고급
def get_score(record_dict, subject_name):
    if subject_name in record_dict:
        return record_dict[subject_name]
    else:
        return "기록 없음"

score_list = {"슬립":20,"미쿠":40,"젤다":15,"키요":50}
print(get_score(score_list,"젤다"))
print(get_score(score_list,"야다"))

def add_playtime(user_dict, user_name, minutes):
    if user_name in user_dict:
        user_dict[user_name] = user_dict[user_name] + minutes
        return user_dict
    else:
        user_dict[user_name] = minutes
        return user_dict

playtime_list = {"미쿠":51,"하은":105,"민수":350,"영자":600,"영철":10}
print(add_playtime(playtime_list,"하은",25))
print(add_playtime(playtime_list,"정숙",55))

def get_level(char_dict, char_name):
    if char_name in char_dict:
        return char_dict[char_name]
    else:
        return 1

level_list = {"메론":1, "사과":10,"딸기":23,"바나나":8}
print(get_level(level_list,"딸기"))
print(get_level(level_list,"복숭아"))
```

(Q7 `is_urgent`는 테스트 호출이 없어서, 채점을 위해 `{"priority":"긴급"}` / `{"priority":"낮음"}` 데이터로 직접 실행 검증함.)
